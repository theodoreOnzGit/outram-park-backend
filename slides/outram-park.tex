\section{What it is}

\begin{frame}{What OUTRAM PARK is}
  A single Cargo workspace housing a pure-Rust reactor simulation suite.
  \vspace{1em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Scale (measured 14 Sep 2026)}
      \begin{itemize}
        \item \textbf{37} member crates
        \item \textbf{1.06 M} lines under \texttt{src/}
        \item \textbf{3{,}104} Rust source files
        \item \textbf{8{,}367} test functions
        \item GPL-3.0 throughout \\ \caveat{(one AGPL-3.0 exception: KOVAN)}
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Design constraints}
      \begin{itemize}
        \item Enums for dispatch --- no \texttt{dyn}, no \texttt{Box}
        \item No lifetime parameters; \texttt{Arc<RwLock<T>>} for shared state
        \item \texttt{uom}-typed physical quantities
        \item Android/Termux \emph{and} \texttt{wasm32} are supported targets
        \item Release-mode builds and tests only
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1em}
  \caveat{Not a wrapper. Every upstream code listed later is an independent Rust
  translation, not an FFI binding.}
\end{frame}

\section{Verification posture}

\begin{frame}{The honest frame: verification is not validation}
  \begin{itemize}
    \item \textbf{Verification} --- is it implemented correctly?
    \item \textbf{Validation} --- does it represent physical reality well enough for its purpose?
  \end{itemize}
  \vspace{0.8em}
  Every crate carries a two-axis maintainer sign-off tracker. \alert{Today, both
  axes read ``not yet manually checked'' on every crate} --- AI-assisted output is
  treated as untrusted draft material until a human reviews it.
  \vspace{0.8em}
  \begin{center}
  \small
  Prototype $\rightarrow$ Unit Tested $\rightarrow$ Integrated $\rightarrow$
  \textbf{Verified} $\rightarrow$ Validated $\rightarrow$ Published
  \end{center}
  \vspace{0.8em}
  \caveat{Intended for education, research, capability building and V\&V only.
  Explicitly not for facility operation, reactor control, licensing decisions,
  safety-critical analysis or emergency response.}
\end{frame}

\begin{frame}{Structural mechanics --- \texttt{farrer-park}}
  Finite-element Analysis for Reactor Reliability, Engineering Response,
  Plasticity And Risk. Ported from MOOSE \cite{moose} and PRISMS.
  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Verified against analytical solutions}
      \begin{itemize}
        \item MMS convergence orders match theory
        \item Patch test at machine precision
        \item Newton convergence order \textbf{2.004}
        \item Crystal plasticity vs analytic Schmid yield: \textbf{1.5e-15} relative
        \item Frame indifference \textbf{7.6e-16}
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Volumetric locking: measured and cured}
      \begin{itemize}
        \item Full integration: MMS order 0.63--1.24
        \item B-bar: \textbf{2.00--2.05}
        \item Plastic collapse vs closed-form limit load: $+23.9\%$ $\rightarrow$ $\mathbf{+1.06\%}$
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \caveat{Shear locking is measured and \alert{not} cured --- 11.25\% too stiff at two
  Quad4 through the depth, 66.7\% at aspect ratio 4. Small strain only: no lattice
  reorientation, no backstress. Verification only; no human V\&V.}
\end{frame}

\section{Known gaps}

\begin{frame}{What is \emph{not} working --- stated deliberately}
  \begin{itemize}
    \item \textbf{916 open issues}, 674 closed. Four P0 defects are
      \texttt{todo!()} panics or arithmetic underflow in live solver paths.
    \item \textbf{No crate has human V\&V sign-off.} Every bookkeeping tracker
      reads INCOMPLETE.
    \item \textbf{The first headless run of \texttt{htgr\_sim\_v1}} (6 Sep 2026)
      showed power overshooting to \textbf{27.8\,MW}, $\sim$2.8$\times$ nominal,
      against a docstring claiming it started near steady state. The docstring had
      been wrong, unnoticed, because a GUI-only simulator cannot be tested.
    \item \textbf{IF97 flash panics above 1073.15\,K} --- too low for an HTGR.
    \item \textbf{\texttt{wasm32} gate proves compilation, not execution.}
      \texttt{thread::spawn} and \texttt{Instant::now} compile and panic at runtime.
    \item \textbf{Six \texttt{extern "C"} calls} to the system libm remain, duplicated
      across three crates.
  \end{itemize}
  \vspace{0.5em}
  \caveat{These are tracked, not hidden. A simulator whose behaviour cannot be
  falsified is not a capability.}
\end{frame}

\section{Where we are going}

\begin{frame}{Near-term: consolidation}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{PETIR} --- core maths crate \\
      \caveat{Polynomials, Equations, Transforms, Integration, Roots}
      \begin{itemize}
        \item \texttt{no\_std} port of GSL \cite{gsl}, Octave and SLATEC \cite{slatec}
        \item \alert{Ported, never AI-generated} --- traceable to a library with a real test suite
        \item Removes the last C from the workspace
        \item Bit-identical transcendentals across platforms
        \item Unblocks NJOY and RAFFLES, neither of which can take the CFD crate
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Why it matters} \cite{trefethen_atap}
      \begin{itemize}
        \item Chebyshev fitting for cross-section representation
        \item Orthogonal-basis regression for surrogate models
        \item $\sim$27k lines of general numerics currently sit inside an OpenFOAM-branded crate
      \end{itemize}
      \vspace{0.5em}
      \textbf{Also near-term}
      \begin{itemize}
        \item Android/Termux build sweep
        \item API dogfooding: \emph{if it is too complex for a small model, it is a bad API}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The through-line}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \textbf{What is genuinely built}
      \begin{itemize}
        \item Nuclear data agreeing with NJOY2016 to 7 figures
        \item MC within 500 pcm of Godiva
        \item TH validated against CIET to 0.2\,\textdegree C
        \item FEM with MMS orders matching theory
        \item A million lines under one licence, one dependency policy, one workspace
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{What makes it trustworthy}
      \begin{itemize}
        \item Every port names its upstream, commit and licence
        \item Every benchmark states methodology \emph{and} measured result
        \item Every gap is a tracked issue, not a silence
        \item AI-written code is untrusted until a human signs it off
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1.2em}
  \begin{center}
    \alert{\textbf{The hard part is not writing the physics.}} \\
    \vspace{0.3em}
    It is being able to prove, later, that it was right.
  \end{center}
\end{frame}
